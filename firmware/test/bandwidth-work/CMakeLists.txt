cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(bandwidth_work C CXX ASM)
pico_sdk_init()
add_executable(bandwidth_work main.c)
pico_generate_pio_header(bandwidth_work ${CMAKE_CURRENT_LIST_DIR}/spi_rx.pio)
target_link_libraries(bandwidth_work pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_set_binary_type(bandwidth_work no_flash)
pico_enable_stdio_usb(bandwidth_work 0)
pico_enable_stdio_uart(bandwidth_work 0)
target_compile_options(bandwidth_work PRIVATE -Wall -Wextra)
pico_add_extra_outputs(bandwidth_work)
