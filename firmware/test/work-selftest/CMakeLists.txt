cmake_minimum_required(VERSION 3.13)
if (NOT PICO_SDK_PATH)
    if (DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK path")
    else()
        message(FATAL_ERROR "PICO_SDK_PATH must point to the installed Pico SDK")
    endif()
endif()
set(PICO_BOARD pico CACHE STRING "dualcore Work core board")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(work_selftest C CXX ASM)
pico_sdk_init()
add_subdirectory(../../harness-target pico_harness)
foreach(target work_selftest work_selftest_ram)
    add_executable(${target} main.c)
    target_link_libraries(${target} pico_harness pico_stdlib hardware_pio hardware_dma hardware_uart)
    if (target STREQUAL "work_selftest_ram")
        pico_set_binary_type(${target} no_flash)
    endif()
    pico_enable_stdio_usb(${target} 0)
    pico_enable_stdio_uart(${target} 0)
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    pico_add_extra_outputs(${target})
endforeach()
